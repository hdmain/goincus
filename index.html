goincus — Incus NAT VPS API

Source: github.com/hdmain/goincus

Debian / Ubuntu (apt)

echo "deb [trusted=yes] https://hdmain.github.io/goincus ./" \
  | sudo tee /etc/apt/sources.list.d/goincus.list
sudo apt update
sudo apt install goincus
sudo goincus init
sudo systemctl enable --now goincus

Fedora / AlmaLinux / CentOS (dnf)

sudo tee /etc/yum.repos.d/goincus.repo <<'REPO'
[goincus]
name=goincus
baseurl=https://hdmain.github.io/goincus/rpm
enabled=1
gpgcheck=0
REPO
sudo dnf install goincus
sudo goincus init
sudo systemctl enable --now goincus
